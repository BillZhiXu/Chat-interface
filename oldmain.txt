# Global variable to store the dataset metadata and sample rows after upload
dataset_metadata = {}
sample_rows = []

# Define request and response models
class QueryRequest(BaseModel):
    prompt: str

# Chart Generation Tool
chart_generation_tool = {
    "type": "function",
    "function": {
        "name": "chart_generation",
        "description": "Generate a Vega-Lite chart specification based on the user's visualization query.",
        "parameters": {
            "type": "object",
            "properties": {
                "prompt": {
                    "type": "string",
                    "description": "The user's request for data visualization, such as 'Create a bar chart showing sales over time'. 'Show a scatter plot of car prices vs. mileage'.",
                },
            },
            "required": ["prompt"],
            "additionalProperties": False,
        }
    }
}

def chart_generation(prompt: str) -> dict:
    """Generates a Vega-Lite chart specification based on the user's prompt."""
    vega_prompt = generate_vega_prompt(dataset_metadata, sample_rows, prompt)
    
    chat_completion = client.chat.completions.create(
        model="gpt-3.5-turbo",
        messages=[
            {"role": "system", "content": "You are a helpful assistant that generates Vega-Lite specifications."},
            {"role": "user", "content": vega_prompt},
        ]
    )
    
    response_content = chat_completion.choices[0].message.content
    return json.loads(response_content)

# Data Analysis Tool
data_analysis_tool = {
    "type": "function",
    "function": {
        "name": "data_analysis",
        "description": "Generate executable data analysis Python code based on the user's analysis query. ",
        "parameters": {
            "type": "object",
            "properties": {
                "query": {
                    "type": "string",
                    "description": "The user's request for data analysis, such as 'Calculate the range of car weights'. 'what is the average price of cars?'",
                },
            },
            "required": ["query"],
            "additionalProperties": False,
        }
    }
}

def data_analysis(query: str) -> str:
    """Generates executable Python code based on the user's analysis query and executes it."""
    # Generate the prompt for the model to convert the query to Python code
    analysis_prompt = generate_analysis_prompt(dataset_metadata, sample_rows, query)
    
    # Call the OpenAI API with the generated prompt
    chat_completion = client.chat.completions.create(
        model="gpt-3.5-turbo",
        messages=[
            {"role": "system", "content": "You are a helpful assistant that generates Python code for data analysis."},
            {"role": "user", "content": analysis_prompt},
        ]
    )
    
    # Extract the generated code from the response
    response_content = chat_completion.choices[0].message.content
    response_content = sanitize_code(response_content)
    print(response_content)
    # Check if the response contains an indication that the request is not related to the provided data
    if "description" in response_content and "not related to the provided data" in response_content:
        print("Request is not related to the provided data.")
        # Return the JSON directly as a string without execution
        return json.loads(response_content)
    # Execute the generated code and return the output
    output = execute_panda_dataframe_code(response_content)
    print("output:", output)
    return json.loads(output)

def sanitize_code(code: str) -> str:
    """Sanitize the generated code by removing markdown code block indicators and stray 'python' keywords."""
    # Remove triple backticks and any leading 'python' markers
    code = code.replace("```", "").strip()
    code = code.replace("python", "").strip()
    return code

# Helper function for safe code execution
def execute_panda_dataframe_code(code: str) -> str:
    old_stdout = sys.stdout
    mystdout = StringIO()
    sys.stdout = mystdout
    try:
        exec(code)
        sys.stdout = old_stdout
        return mystdout.getvalue()
    except Exception as e:
        sys.stdout = old_stdout
        return repr(e)

# Define tools and tool map
tools = [chart_generation_tool, data_analysis_tool]

tool_map = {
    "chart_generation": chart_generation,
    "data_analysis": data_analysis
}


def generate_analysis_prompt(data_info, sample_data, user_query):
    analysis_prompt = f"""
You are an expert data analysis that uses all the data to generate analysis code. The user has provided the following query:: '{user_query}'.

The full dataset is already loaded into a Pandas DataFrame named `df`, containing all the data points provided by the user. The dataset includes the following columns and data types:
{json.dumps(data_info, indent=2)}

Here are the dataset's structure (MAKE SURE all col have 30 values) SET this to equal to the data dictionary(Dictionary of Lists):
{json.dumps(sample_data, indent=2)}

If the user's request contain column variables NOT PROVIDED in the data, provide the following response:
{{
  "description": "It seems that your request is not related to the provided data."
}}

If the user's request relates to data analysis (pay close attention to words refering to columns of the dataset), generate Python code to perform the analysis. Ensure that:
1. Please make sure that the code won't rasie any error and will run without any issue. ie. not SyntaxError, ValueError, etc.
2. The code includes necessary imports if required (e.g., `import pandas as pd` 'import json).
3. Use a data dictionary (Dictionary of Lists) to store the ALL the data provided (ENSURE that all arrays are of the SAME length). then convert it to a pandas DataFrame. 
4. The code performs only the requested operation and prints the result so it can be captured.
5. Use the `print()` function to output the result explicitly with 'json.dumps()' and in f-string (formatted string literal) format. 
The print should output in the following JSON format with DOUBLE quotes, up to two decimal points and ONLY "description" as key:
 for example:
 print(json.dumps({{"description": "The range of car weights is 100"}}))
 print(json.dumps({{"description": "average MPG from Europe is 25, average MPG from Japan is 30, and average MPG from US is 20"}}))
6. The code should not include any nonexecutable content (e.g., comments, markdown, or explanations).

"""

    return analysis_prompt


# Function to generate the prompt for Vega-Lite, including sample rows
def generate_vega_prompt(data_info, sample_data, user_prompt):
    vega_prompt = f"""
    IMPORTANT: You must respond with ONLY a JSON object. No other text, no markdown, no explanations outside the JSON.

You are an expert in data visualization. The user has provided the following query: '{user_prompt}'.

You have access to a dataset with the following columns and associated information:
{json.dumps(data_info, indent=2)}
 Here are dataset datapoints:
{json.dumps(sample_data, indent=2)}

If the user's request relates to creating a data visualization (pay close attention to words refering to columns of the dataset), generate a JSON response that contains:
1. "description": A concise explanation of the chart that you've created.
2. "specification": A full Vega-Lite specification, including:
   - "$schema": The URL for the Vega-Lite schema.
   - "data": A JSON object that includes the sample data points under the "values" key.
   - "mark": The type of visualization to be generated (e.g., "bar", "line", "point").
   - "encoding": Details on how the data fields are mapped to the axes and other visual elements (x-axis, y-axis, color, etc.).

If the user's request contain variables UNRELATED or not PRESENT in the data, provide the following response:
{{
  "description": "It seems that your request is not related to the provided data."
}}

Here is an example of the Vega-Lite specification format:
{{
  "$schema": "https://vega.github.io/schema/vega-lite/v5.json",
  "data": {{
    "values": [
      {{"category": "A", "group": "x", "value": 0.1}},
      {{"category": "A", "group": "y", "value": 0.6}},
      {{"category": "A", "group": "z", "value": 0.9}},
      {{"category": "B", "group": "x", "value": 0.7}},
      {{"category": "B", "group": "y", "value": 0.2}},
      {{"category": "B", "group": "z", "value": 1.1}},
      {{"category": "C", "group": "x", "value": 0.6}},
      {{"category": "C", "group": "y", "value": 0.1}},
      {{"category": "C", "group": "z", "value": 0.2}}
    ]
  }},
  "mark": "bar",
  "encoding": {{
    "x": {{"field": "category"}},
    "y": {{"field": "value", "type": "quantitative"}},
    "xOffset": {{"field": "group"}},
    "color": {{"field": "group"}}
  }}
}}
"""

    return vega_prompt

# Endpoint to handle CSV file upload and extract metadata and sample rows
@app.post("/upload-data")
async def upload_data(file: UploadFile = File(...)):
    try:
        # Load the CSV file into a pandas DataFrame
        df = pd.read_csv(file.file)

        # Extract column names and data types
        data_info = {
            "columns": df.columns.tolist(),
            "types": [str(df[col].dtype) for col in df.columns]
        }

        # # Get 50 sample rows
        sample_data = df.sample(n=30, random_state=1).to_dict(orient='records')

        # Store the dataset metadata and the sample rows in global variables
        global dataset_metadata, sample_rows
        dataset_metadata = data_info
        sample_rows = sample_data  # Store the sample rows for use in the query

        return {
            "message": "Data uploaded successfully.",
            "data_info": data_info,
            "sample_rows": sample_data  # Include the all rows in the response
        }

    except Exception as e:
        raise HTTPException(status_code=400, detail="Error processing file: " + str(e))


@app.post("/query")
async def query_openai(request: QueryRequest):
    try:
        global dataset_metadata, sample_rows
        if not dataset_metadata or not sample_rows:
            raise HTTPException(status_code=400, detail="No dataset uploaded yet. Please upload a dataset first.")

        # Prepare the initial prompt and tools
        prompt = request.prompt
        messages = [
            {"role": "system", "content": "You are a helpful assistant. Use the supplied tools to assist the user. If the request is a Chart and analysis query, use both tools."},
            {"role": "user", "content": prompt}
        ]

        # Initialize placeholders for results from each tool
        result_data = {
            "specification": None,
            "description": None
        }

        max_iterations = 3  # Max number of tool-call iterations
        for i in range(max_iterations):
            print("iteration:", i+1)
            response = client.chat.completions.create(
                model="gpt-3.5-turbo-1106",
                messages=messages,
                tools=tools,
            )

            # If the response has final content, return it
            print(response.choices[0].message)
            if response.choices[0].message.content is not None:
                print("Final response:", response.choices[0].message.content)
                return response.choices[0].message.content

            # If there are tool calls, proceed with each one
            if response.choices[0].message.tool_calls is not None:
                messages.append(response.choices[0].message)  # Add the assistant's tool call response

                # Iterate over all tool calls
                for tool_call in response.choices[0].message.tool_calls:
                    print("calling:", tool_call.function.name, "with", tool_call.function.arguments)

                    # Parse arguments and call the appropriate function
                    arguments = json.loads(tool_call.function.arguments)
                    function_to_call = tool_map[tool_call.function.name]
                    result = function_to_call(**arguments)

                    # Determine if the tool call is for chart generation or data analysis
                    if tool_call.function.name == "chart_generation":
                        result_data["specification"] = result.get("specification")
                    elif tool_call.function.name == "data_analysis":
                        result_data["description"] = result.get("description")

                    # Create a message with the tool's result
                    result_content = json.dumps({**arguments, "result": result})
                    function_call_result_message = {
                        "role": "tool",
                        "content": result_content,
                        "tool_call_id": tool_call.id,
                    }
                    print("Action result:", result_content)

                    # Append the tool result message back to messages for the assistant to use
                    messages.append(function_call_result_message)

            # Check if we have the necessary results
            if result_data["specification"] and result_data["description"]:
                # If both chart and analysis are provided, return both
                return result_data
            elif result_data["description"] and not result_data["specification"]:
                # If only data analysis is needed, return only description
                return {"description": result_data["description"]}

        # If max iterations are reached and results are incomplete
        print("Max iterations reached")
        return {"error": "The tool agent could not complete the task in the given time. Please try again."}

    except Exception as e:
        logger.error("Error processing the query: %s", str(e))
        raise HTTPException(status_code=500, detail=str(e))
